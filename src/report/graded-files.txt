# One graded-input list for build.rs and report::rules.
#
# `file` and `dir` entries are compiled into the binary. build.rs hashes
# them at build time and emits SDB_COMPILED_RULES_DIGEST; changing any of
# them (or rebuilding after a change) moves the stamp.
#
# `runtime-dir` entries are read from SDB_HOME at run time. rules.rs hashes
# them when it computes the fingerprint, so a deployment with different
# stage prompts gets a different stamp without rebuilding.

file src/campaign.rs
file src/scoring.rs

dir src/verifier

dir src/cost
dir src/sandbox

runtime-dir tasks
